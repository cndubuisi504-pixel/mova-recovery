import { createPublicClient, http, isAddress, getAddress, zeroAddress, encodeFunctionData, formatEther } from 'viem';
import { mainnet } from 'viem/chains';
import { createHash } from 'node:crypto';

// Research allowlist. A database row cannot enable a new executable method.
export const protocol = Object.freeze({
  id: 'etherdelta-eth', name: 'EtherDelta', chainId: 1, asset: 'ETH',
  contract: '0x8d12A197cB00D4747a1fe03395095ce2A5CC6819',
  destinationType: 'owner_only', status: 'research_only',
  runtimeSha256: '7d73bf3b09322a85e37a0a34ccb35458bc6ee38999394f7819ef80db524fb8aa',
});
export const abi = [
  { type:'function', name:'balanceOf', stateMutability:'view', inputs:[{name:'token',type:'address'},{name:'user',type:'address'}], outputs:[{type:'uint256'}] },
  { type:'function', name:'withdraw', stateMutability:'nonpayable', inputs:[{name:'amount',type:'uint256'}], outputs:[] },
  { type:'event', name:'Withdraw', inputs:[{name:'token',type:'address',indexed:false},{name:'user',type:'address',indexed:false},{name:'amount',type:'uint256',indexed:false},{name:'balance',type:'uint256',indexed:false}] },
];
export class RecoveryError extends Error {
  constructor(code, message, status=400) { super(message); this.code=code; this.status=status; }
}
export function validateWallet(input) {
  if (typeof input !== 'string' || !isAddress(input, { strict:true }) || input.toLowerCase() === zeroAddress)
    throw new RecoveryError('invalid_address','Enter a valid, nonzero Ethereum address. Check the address and its capitalization.');
  return getAddress(input);
}
export function onlyWallet(body) {
  if (!body || typeof body !== 'object' || Object.keys(body).some(k=>k!=='wallet'))
    throw new RecoveryError('unsupported_request','This checker only accepts a wallet address.');
  return validateWallet(body.wallet);
}
export function buildResearchWithdrawal(wallet, amount) {
  wallet=validateWallet(wallet);
  if (typeof amount !== 'bigint' || amount <= 0n || amount >= 2n**256n) throw new RecoveryError('position_unavailable','No positive ETH position is available to check.');
  return Object.freeze({ account:wallet, to:protocol.contract, data:encodeFunctionData({abi,functionName:'withdraw',args:[amount]}), value:0n });
}
export function makeClient(url) { return createPublicClient({chain:mainnet, transport:http(url, {timeout:12000,retryCount:1})}); }
export async function discoverPosition(client, walletInput) {
  const wallet=validateWallet(walletInput);
  if (await client.getChainId() !== 1) throw new RecoveryError('wrong_rpc_network','The checker is not connected to Ethereum Mainnet. Please try later.',503);
  const block=await client.getBlock({blockTag:'latest'});
  const code=await client.getCode({address:protocol.contract,blockNumber:block.number});
  if (!code || code==='0x' || createHash('sha256').update(Buffer.from(code.slice(2),'hex')).digest('hex')!==protocol.runtimeSha256)
    throw new RecoveryError('protocol_changed','The protocol contract could not be matched to our research record. Checking is paused.',503);
  const amount=await client.readContract({address:protocol.contract,abi,functionName:'balanceOf',args:[zeroAddress,wallet],blockNumber:block.number});
  if ((await client.getBlock({blockNumber:block.number})).hash!==block.hash) throw new RecoveryError('state_changed','Network state changed during the check. Please scan again.',409);
  return { wallet, amountWei:amount.toString(), amount:formatEther(amount), protocol, blockNumber:block.number.toString(), blockHash:block.hash, observedAt:new Date().toISOString(), state:amount>0n?'position_found':'no_position', signingEnabled:false };
}
export async function simulateRecovery(client,wallet) {
  // Always rediscover. Never accept transaction fields or cached amount from a browser.
  const position=await discoverPosition(client,wallet);
  const transaction=buildResearchWithdrawal(position.wallet,BigInt(position.amountWei));
  const blockNumber=BigInt(position.blockNumber);
  try {
    await client.call({...transaction,blockNumber});
  } catch {
    throw new RecoveryError('simulation_failed',"The withdrawal couldn't be simulated successfully right now. No funds have moved.",422);
  }
  if ((await client.getBlock({blockNumber})).hash!==position.blockHash) throw new RecoveryError('state_changed','Network state changed. Run the check again.',409);
  let gasEstimate=null;
  try { gasEstimate=(await client.estimateGas({...transaction,blockNumber})).toString(); } catch { /* A gas estimate is optional and cannot turn a failed simulation into a pass. */ }
  return {...position,simulation:'passed',gasEstimate,signingEnabled:false,transaction:{from:position.wallet,to:transaction.to,data:transaction.data,value:'0x0',chainId:1},note:'Read-only eth_call at the recorded block. This is not a completed recovery, proof of wallet control, security audit, or guarantee of future execution.'};
}


function reply(body,status=200,headers={}){
 return new Response(JSON.stringify(body),{status,headers:{'Content-Type':'application/json; charset=utf-8','Cache-Control':'no-store','X-Content-Type-Options':'nosniff',...headers}});
}
export default async function handler(request){
 const path=new URL(request.url).pathname.replace(/\/$/,'');
 const action=path.split('/').pop();
 if(!['config','scan','simulate'].includes(action))return reply({error:'not_found',message:'This action is not available in the research checker.'},404);
 const allowed=action==='config'?'GET':'POST';
 if(request.method!==allowed)return reply({error:'method_not_allowed',message:'This request method is not supported.'},405,{Allow:allowed});
 if(action==='config')return reply({chainId:1,signingEnabled:false,mode:'read_only_research',rpcConfigured:!!process.env.ETHEREUM_RPC_URL,walletConnectProjectId:process.env.WALLETCONNECT_PROJECT_ID||null});
 try{
  const declared=Number(request.headers.get('content-length'));
  if(declared>2048)return reply({error:'request_too_large',message:'Only a wallet address is needed.'},413);
  const text=await request.text();
  if(new TextEncoder().encode(text).byteLength>2048)return reply({error:'request_too_large',message:'Only a wallet address is needed.'},413);
  let body;
  try{body=JSON.parse(text);}catch{return reply({error:'invalid_request',message:'The request could not be read. Please try again.'},400);}
  const wallet=onlyWallet(body);
  const client=makeClient(process.env.ETHEREUM_RPC_URL||'https://ethereum-rpc.publicnode.com');
  const result=action==='scan'?await discoverPosition(client,wallet):await simulateRecovery(client,wallet);
  return reply(result);
 }catch(error){
  if(error instanceof RecoveryError)return reply({error:error.code,message:error.message},error.status);
  return reply({error:'service_unavailable',message:'We couldn’t complete the check. The network service may be unavailable. Please try again shortly.'},503);
 }
}
