export async function simulateStader(client,walletInput,kind,requestKey){
 if(!['request','claim'].includes(kind))throw new RecoveryError('unsupported_request','Choose a supported Stader check.');
 const position=await discoverStader(client,walletInput);let transaction,amountWei=null,stage;
 if(kind==='request'){
  const balance=BigInt(position.amountWei);if(balance===0n)throw new RecoveryError('position_unavailable','No MaticX balance is available for a new request. Existing requests are checked separately.');
  if(position.stader.instant)throw new RecoveryError('mode_changed','Stader is in instant-redemption mode. That path needs separate review; no signing is enabled.',409);
  if(position.stader.paused)throw new RecoveryError('withdrawals_paused','New Stader withdrawal requests are currently paused. Existing claims may follow different rules.',409);
  const amount=balance<1000000000000000n?balance:1000000000000000n;amountWei=amount.toString();stage='request_only';transaction={account:position.wallet,to:staderProject.contract,data:encodeFunctionData({abi:staderAbi,functionName:'requestWithdraw',args:[amount]}),value:0n};
 }else{
  if(typeof requestKey!=='string'||!/^0x[0-9a-fA-F]{64}$/.test(requestKey))throw new RecoveryError('invalid_request','Select a current withdrawal request.');
  const item=position.stader.requests.find(r=>r.key===requestKey);if(!item)throw new RecoveryError('request_changed','The request changed or was already claimed. Refresh the position.',409);
  if(item.state!=='epoch_eligible')throw new RecoveryError('not_ready','The required checkpoint epochs have not elapsed. No claim was submitted.',409);
  stage='claim_only';transaction={account:position.wallet,to:staderProject.contract,data:encodeFunctionData({abi:staderAbi,functionName:'claimWithdrawal',args:[BigInt(item.index)]}),value:0n};
 }
 let gasEstimate=null;try{await client.call({...transaction,blockNumber:BigInt(position.blockNumber)});}catch{throw new RecoveryError('simulation_failed','This Stader action did not pass its read-only simulation. No transaction was submitted.',422);}
 try{gasEstimate=(await client.estimateGas({...transaction,blockNumber:BigInt(position.blockNumber)})).toString();}catch{}
 if((await client.getBlock({blockNumber:BigInt(position.blockNumber)})).hash!==position.blockHash)throw new RecoveryError('state_changed','The verification block changed. Check again.',409);
 return {simulation:'passed',stage,amountWei,requestKey:kind==='claim'?requestKey:null,gasEstimate,blockNumber:position.blockNumber,blockHash:position.blockHash,signingEnabled:false,note:kind==='request'?'A small request was simulated, not submitted. Requesting is not recovering; POL arrives only after an eligible claim.':'The freshly located claim was simulated, not submitted. Epoch eligibility alone is not proof of a successful payment.'};
}

export async function readSiliconService(path){

}

function siliconProofRoot(leaf,index,siblings){if(!/^0x[0-9a-fA-F]{64}$/.test(leaf)||!Array.isArray(siblings)||siblings.length!==32||siblings.some(s=>!/^0x[0-9a-fA-F]{64}$/.test(s)))throw new RecoveryError('invalid_proof','Silicon returned an invalid proof.',503);let current=leaf,n=quantity(index);if(n>=2n**32n)throw new RecoveryError('invalid_proof','The exit index is out of range.',503);for(const sibling of siblings){current=keccak256((n&1n)===0n?concatHex([current,sibling]):concatHex([sibling,current]));n>>=1n;}return current;}
